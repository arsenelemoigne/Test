rlcal --name rcs --start soup:ca1 --data-seed 31 --smoke

# 2. on ca1 (clean, unseen +48.6% / -1.8): calibrated cuts at 1-80% of the thinking -> DPO 3 seeds -> RL 120 steps in blocks of 20
rlcal --name rc1 --start soup:ca1 --data-seed 31 --gen-questions 1000 --kod-questions 300 --floor -1.5
# smoke test of the RL-only path again, after the fix
rlcal --name rcs4 --start models/rep4-blend50 --data-seed 32 --rl-only --smoke --rl-block 1

rlcal --name rc1 --start soup:ca1 --data-seed 31 --gen-questions 1000 --kod-questions 300 --floor -1.5 --rl-block 20
# 3. (shared with runtime 1: whichever runtime gets here first runs it) RL only, on the round 4 50/50 blend (unseen +57.3% / -2.4, MBPP grey zone): the closest to >55% / -2
# rc1 with the stopping rule per type (23 Oct, 03:50): the calibration showed ca1's confidence is too poorly
# calibrated for math, code, tools and instructions, so those types are cut where the 20 re-answers are all right
rlcal --name rc1 --start soup:ca1 --data-seed 31 --gen-questions 1000 --kod-questions 300 --floor -1.5 --rl-block 20 --seeds 0 1 2
rlcal --name rc1 --start soup:ca1 --data-seed 31 --gen-questions 1000 --kod-questions 300 --floor -1.5 --rl-block 20 --seeds 0 1 2 --rl-alpha 0.15

# 23 Oct ~09:00: RL stopped after 3 blocks (saving 55 -> 54 -> 51 -> 48.5% on the choosing set, accuracy flat, thinking
# rising to 400-500 tokens on the repeated questions): choose among the DPO soup and blocks 1-3, report x8
rlcal --name rc1 --start soup:ca1 --data-seed 31 --gen-questions 1000 --kod-questions 300 --floor -1.5 --rl-block 20 --seeds 0 1 2 --rl-alpha 0.15 --rl-steps 60
# rr4 on fresh questions (each seen about once over 120 RL steps), 23 Oct ~08:15
shared rlcal --name rr4 --start models/rep4-blend50 --data-seed 32 --gen-questions 1000 --kod-questions 300 --floor -1.5 --rl-only --rl-steps 120 --rl-fresh-seed 41 --rl-alpha 0.3
